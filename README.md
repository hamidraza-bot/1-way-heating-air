async function run(act){
  const [k,arg]=act.split(/:(.*)/s);
  switch(k){

function startRequest(preset){
  chat.data=blank();chat.editing=false;chat.emergencyShown=false;chat.id=newId();
  if(preset)chat.data.service=preset;
  ask('name');
}
function ask(step){
  chat.lastStep=step;
  switch(step){
    case 'name':chat.mode='name';return botSay("What's your name?");
    case 'phone':chat.mode='phone';return botSay("What's the best phone number to reach you?");
    case 'service':chat.mode='off';return botSay('What type of HVAC service do you need?',SERVICES.map(s=>({label:s,act:'svc:'+s})));
    case 'message':chat.mode='message';return botSay("Tell us briefly what's happening.");
    case 'urgency':chat.mode='off';return botSay('How urgent is this?',URGENCY.map(s=>({label:s,act:'urg:'+s})));
    case 'summary':return summary();
  }
}
const ORDER=['name','phone','service','message','urgency'];
function advance(step){
  if(chat.editing){chat.editing=false;return summary()}
  for(let i=ORDER.indexOf(step)+1;i<ORDER.length;i++){const s=ORDER[i];
    if(s==='service'&&chat.data.service)continue;
    if(s==='urgency'&&chat.data.urgency)continue;
    return ask(s)}
  return summary();
}
function chooseService(s){chat.data.service=s;

  if(em&&!chat.emergencyShown){chat.emergencyShown=true;chat.mode='off';chat.pending=()=>advance(step);return emergencyMsg()}

function handleText(raw){
  const text=raw.trim();if(!text)return;
  const m=chat.mode,d=chat.data;
  userSay(text);
  if(m==='name'){
    if(text.length<2||/\d{3,}/.test(text)){return botSay('Please enter your full name.')}
    d.name=text;return advance('name')}
  if(m==='phone'){
    if(digits(text).length<10){return botSay("That doesn't look like a complete phone number. Please include the area code.")}
    d.phone=text;return advance('phone')}
  if(m==='message'){
    if(text.length<3){return botSay('Please tell us a little more about what’s happening.')}
    d.message=text;return advance('message')}
  const hit=MATCH.find(([,re])=>re.test(text.toLowerCase()));
  return faq(hit?hit[0]:'unknown');
}
